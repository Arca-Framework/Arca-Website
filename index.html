Read the API 
-- pay every on-duty medic a bonus from the EMS bank account
for _, src in ipairs(exports.arca_core:GetOnDuty('ambulance')) do
    if exports.arca_bank:RemoveJobMoney('ambulance', 250, 'Bonus') then
        Arca.Functions.GetPlayer(src).AddMoney('bank', 250, 'EMS bonus')
        Arca.Notify(src, 'Bonus received: $250', 'success')
    end
end
-- a third-eye option with a progress bar and a server check
exports.arca_target:addModel({ 'prop_atm_01' }, {{
    name = 'rob_atm', label = 'Hack ATM', icon = 'fa-solid fa-laptop-code',
    onSelect = function()
        if Arca.Progress({ label = 'Hacking', duration = 8000 }) then
            local payout = Arca.Callback.Await('heist:atm')
            Arca.Notify(('You got $%d'):format(payout), 'success')
        end
    end,
}})
-- a usable item, removed only if the action finished
Arca.Functions.CreateUseableItem('energydrink', function(source, item)
    if not Arca.Callback.Await('drink:progress', source, 10000) then return end
    if exports.arca_inventory:RemoveItem(source, item.name, 1, item.slot) then
        TriggerClientEvent('drink:boost', source)
    end
end)

Get Arca

Install with txAdmin

A fresh Arca server in a few minutes. You need Windows or Linux, MariaDB and a FiveM licence key.

  1. 1

    Install MariaDB

    Version 10.6 or newer. Note the root password you set.

  2. 2

    Download FXServer

    Get the recommended artifacts, run FXServer and open txAdmin in your browser.

  3. 3

    Use the Arca recipe

    In the setup choose Popular Recipes → Remote URL Template and paste:

    https://raw.githubusercontent.com/Arca-Framework/txAdmin-recipe/main/recipe.yaml
  4. 4

    Start and join

    txAdmin downloads everything and imports the database. Connect, create a character, done.
